       rm -rf Versionfile
fi
touch Versionfile
echo "capp-${version}" >> Versionfile

git add ./*

git push origin ${code_branch}

tagName="capp-${version}"

# 磁盘挂载命令
、、、
pvcreate /dev/vdb && vgcreate apps_data /dev/vdb && lvcreate -l 100%FREE -n lv_apps_data apps_data && mkfs.xfs -isize=2048 /dev/mapper/apps_data-lv_apps_data && mkdir /apps -p && mount /dev/mapper/apps_data-lv_apps_data /apps && echo '/dev/mapper/apps_data-lv_apps_data /apps xfs defaults        0 0' >> /etc/fstab
chown -R apps:apps /apps
